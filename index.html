// ================================================================ // ServiceM8 Timesheet Proxy — Google Apps Script // Version: 4.0 — Clean final version // ================================================================ // SETUP: // 1. Replace YOUR_API_KEY_HERE with your ServiceM8 API key // 2. Replace YOUR_GOOGLE_SHEET_ID_HERE with your Google Sheet ID // ================================================================ var SM8_API_KEY = "YOUR_API_KEY_HERE"; var SHEET_ID = "YOUR_GOOGLE_SHEET_ID_HERE"; var ALLOWED_ROLES = [ "63fd7881-fe93-477c-9609-23b7fe97daab", "a4d555f9-5bef-4976-87eb-23b7f384ff6b", "ebf88fbe-e6e5-4ad0-bca4-23b7f36f587b", "e814aa90-3a80-4c19-92e7-23b7f0c4f00b" ]; // ---------------------------------------------------------------- // ServiceM8 helpers // ---------------------------------------------------------------- function sm8Get(endpoint) { var res = UrlFetchApp.fetch("https://api.servicem8.com/api_1.0/" + endpoint, { headers: { "X-API-Key": SM8_API_KEY, "Accept": "application/json" }, muteHttpExceptions: true }); if (res.getResponseCode() !== 200) throw new Error("SM8 " + res.getResponseCode() + ": " + res.getContentText().slice(0, 120)); return JSON.parse(res.getContentText()); } function sm8Post(endpoint, payload) { var res = UrlFetchApp.fetch("https://api.servicem8.com/api_1.0/" + endpoint, { method: "post", contentType: "application/json", headers: { "X-API-Key": SM8_API_KEY }, payload: JSON.stringify(payload), muteHttpExceptions: true }); return { code: res.getResponseCode(), body: res.getContentText() }; } function sm8Delete(endpoint) { UrlFetchApp.fetch("https://api.servicem8.com/api_1.0/" + endpoint, { method: "post", contentType: "application/json", headers: { "X-API-Key": SM8_API_KEY }, payload: JSON.stringify({ active: 0 }), muteHttpExceptions: true }); } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } // ---------------------------------------------------------------- // Google Sheet helpers // ---------------------------------------------------------------- function getOrCreateSheet(name, headers) { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); var r = sheet.getRange(1, 1, 1, headers.length); r.setBackground("#1A1917"); r.setFontColor("#FFFFFF"); r.setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } // Save submission — stores full data JSON + activity/note UUIDs for clean editing function saveSubmission(staffUuid, weekEnding, staffName, data, uuids) { var headers = ["Key","Staff UUID","Staff Name","Week Ending","Submitted At","Data JSON","Activity UUIDs","Note UUIDs"]; var sheet = getOrCreateSheet("Submissions", headers); var key = staffUuid + "_" + weekEnding; var rows = sheet.getDataRange().getValues(); var now = new Date().toLocaleString("en-AU"); var record = [key, staffUuid, staffName, weekEnding, now, JSON.stringify(data), JSON.stringify(uuids.activities || []), JSON.stringify(uuids.notes || [])]; for (var i = 1; i < rows.length; i++) { if (rows[i][0] === key) { sheet.getRange(i + 1, 1, 1, headers.length).setValues([record]); return; } } sheet.appendRow(record); } // Load previous submission — returns { data, activityUuids, noteUuids } or null function loadSubmission(staffUuid, weekEnding) { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheetByName("Submissions"); if (!sheet) return null; var key = staffUuid + "_" + weekEnding; var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] === key) { try { return { data: JSON.parse(rows[i][5]), activityUuids: JSON.parse(rows[i][6] || "[]"), noteUuids: JSON.parse(rows[i][7] || "[]") }; } catch(e) { return null; } } } return null; } // Delete only the exact activities and notes we previously created function deletePreviousEntries(activityUuids, noteUuids) { (activityUuids || []).forEach(function(uuid) { try { sm8Delete("jobactivity/" + uuid + ".json"); } catch(e) {} }); (noteUuids || []).forEach(function(uuid) { try { sm8Delete("note/" + uuid + ".json"); } catch(e) {} }); } // Write to All Timesheets tab — replaces existing rows for this staff + week function appendToTimesheetLog(data, weekEnding, staffName) { var headers = ["Week Ending","Staff Name","Day","Date","Job Number", "Start","Finish","Travel (min)","Break (min)","Net Hours","Job Notes"]; var sheet = getOrCreateSheet("All Timesheets", headers); var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (String(rows[i][0]) === String(weekEnding) && String(rows[i][1]) === String(staffName)) { sheet.deleteRow(i + 1); } } var newRows = []; data.days.forEach(function(day) { if (day.no_work || !day.jobs || day.jobs.length === 0) return; day.jobs.forEach(function(job) { if (!job.start_time && !job.job_number) return; newRows.push([weekEnding, staffName, day.day, day.date || "", job.job_number || "", job.start_time || "", job.finish_time || "", job.travel_minutes || 0, job.break_minutes || 0, job.net_hours || 0, job.notes || ""]); }); }); if (newRows.length > 0) { sheet.insertRowsAfter(1, newRows.length); sheet.getRange(2, 1, newRows.length, headers.length).setValues(newRows); } updateWeeklyTotals(data, weekEnding, staffName); } function updateWeeklyTotals(data, weekEnding, staffName) { var headers = ["Week Ending","Staff Name", "Mon","Tue","Wed","Thu","Fri","Sat","Sun", "Total Hours","Total Travel (min)","Total Break (min)","Jobs"]; var sheet = getOrCreateSheet("Weekly Totals", headers); var DAYS = ["Monday","Tuesday","Wednesday","Thursday","Friday","Saturday","Sunday"]; var dayHours = {"Monday":0,"Tuesday":0,"Wednesday":0,"Thursday":0,"Friday":0,"Saturday":0,"Sunday":0}; var totalHours = 0, totalTravel = 0, totalBreak = 0, jobCount = 0; data.days.forEach(function(day) { if (day.no_work || !day.jobs) return; day.jobs.forEach(function(job) { if (!job.start_time && !job.job_number) return; var h = parseFloat(job.net_hours) || 0; dayHours[day.day] = (dayHours[day.day] || 0) + h; totalHours += h; totalTravel += parseInt(job.travel_minutes) || 0; totalBreak += parseInt(job.break_minutes) || 0; jobCount++; }); }); var rows = sheet.getDataRange().getValues(); var existRow = -1; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(weekEnding) && String(rows[i][1]) === String(staffName)) { existRow = i + 1; break; } } var row = [weekEnding, staffName, dayHours["Monday"], dayHours["Tuesday"], dayHours["Wednesday"], dayHours["Thursday"], dayHours["Friday"], dayHours["Saturday"], dayHours["Sunday"], parseFloat(totalHours.toFixed(2)), totalTravel, totalBreak, jobCount]; if (existRow > 0) { sheet.getRange(existRow, 1, 1, headers.length).setValues([row]); } else { sheet.insertRowAfter(1); sheet.getRange(2, 1, 1, headers.length).setValues([row]); } } // ---------------------------------------------------------------- // GET handler // ---------------------------------------------------------------- function doGet(e) { try { var action = e && e.parameter && e.parameter.action; // Return staff list if (action === "staff") { var staff = sm8Get("staff.json") .filter(function(s) { return s.active == 1 && ALLOWED_ROLES.indexOf(s.security_role_uuid) !== -1; }) .map(function(s) { return { uuid: s.uuid, name: (s.first + " " + s.last).trim() }; }) .sort(function(a,b) { return a.name.localeCompare(b.name); }); return jsonOut({ success: true, staff: staff }); } // Return schedule — previous submission or empty days if (action === "schedule") { var staffUuid = e.parameter.staff_uuid; var weekEnding = e.parameter.week_ending; if (!staffUuid || !weekEnding) return jsonOut({ success: false, message: "Missing parameters" }); // Check Google Sheet for a previous submission first var previous = loadSubmission(staffUuid, weekEnding); if (previous && previous.data && previous.data.days) { return jsonOut({ success: true, schedule: previous.data.days, is_edit: true }); } // No previous submission — return empty days var friday = new Date(weekEnding + "T00:00:00"); var monday = new Date(friday); monday.setDate(friday.getDate() - 4); var DAYS = ["Monday","Tuesday","Wednesday","Thursday","Friday","Saturday","Sunday"]; var result = DAYS.map(function(day, i) { var d = new Date(monday); d.setDate(monday.getDate() + i); return { day: day, date: d.toLocaleDateString("en-AU", { day: "2-digit", month: "2-digit" }), jobs: [] }; }); return jsonOut({ success: true, schedule: result, is_edit: false }); } return jsonOut({ success: true, status: "Timesheet proxy is running." }); } catch(err) { return jsonOut({ success: false, message: err.message }); } } // ---------------------------------------------------------------- // POST handler // ---------------------------------------------------------------- function doPost(e) { try { var data = JSON.parse(e.postData.contents); var staffName = data.staff_name || "Unknown"; var staffUuid = data.staff_uuid || ""; var weekEnding = data.week_ending || ""; var days = data.days || []; var isEdit = data.is_edit || false; var submitted = []; var errors = []; var createdActivities = []; var createdNotes = []; // If editing — delete only our previously created entries if (isEdit) { try { var prev = loadSubmission(staffUuid, weekEnding); if (prev) deletePreviousEntries(prev.activityUuids, prev.noteUuids); } catch(delErr) { errors.push("Cleanup warning: " + delErr.message); } } var DAYS = ["Monday","Tuesday","Wednesday","Thursday","Friday","Saturday","Sunday"]; for (var i = 0; i < days.length; i++) { var day = days[i]; if (day.no_work || !day.jobs || day.jobs.length === 0) continue; for (var j = 0; j < day.jobs.length; j++) { var job = day.jobs[j]; if (!job.job_uuid || job.job_uuid.length < 10) continue; // Calculate actual date for this day of the week var friday = new Date(weekEnding + "T00:00:00"); var dayIndex = DAYS.indexOf(day.day); var jobDate = new Date(friday); jobDate.setDate(friday.getDate() - (4 - dayIndex)); var dateStr = jobDate.getFullYear() + "-" + String(jobDate.getMonth() + 1).padStart(2, "0") + "-" + String(jobDate.getDate()).padStart(2, "0"); // 1. Post job activity to ServiceM8 var actResult = sm8Post("jobactivity.json", { job_uuid: job.job_uuid, staff_uuid: staffUuid, start_date: dateStr + " " + (job.start_time || "08:00") + ":00", end_date: dateStr + " " + (job.finish_time || "08:00") + ":00", activity_was_recorded: "1", activity_was_scheduled: "0", travel_time_in_seconds: (job.travel_minutes || 0) * 60, active: 1 }); if (actResult.code === 200 || actResult.code === 201) { submitted.push("Job " + (job.job_number || job.job_uuid.slice(0, 8)) + " on " + day.day); try { var ab = JSON.parse(actResult.body); if (ab.uuid) createdActivities.push(ab.uuid); } catch(ex) {} } else { errors.push("Job " + (job.job_number || "?") + ": " + actResult.body.slice(0, 100)); continue; } // 2. Post diary note to ServiceM8 var noteLines = [ "=== TIMESHEET ENTRY ===", "Staff: " + staffName, "Week: " + weekEnding, "Date: " + day.day + " " + (day.date || dateStr), "Start: " + (job.start_time || "—"), "Finish: " + (job.finish_time || "—"), "Travel: " + (job.travel_minutes || 0) + " min", "Break: " + (job.break_minutes || 0) + " min", "Net hours: " + (job.net_hours || 0).toFixed(2) + " hrs", job.notes ? "Job notes: " + job.notes : "" ].filter(function(s) { return s !== ""; }).join("\n"); var noteResult = sm8Post("note.json", { related_object: "job", related_object_uuid: job.job_uuid, note: noteLines, active: 1 }); try { var nb = JSON.parse(noteResult.body); if (nb.uuid) createdNotes.push(nb.uuid); } catch(ex) {} } } // 3. Save to Google Sheet try { saveSubmission(staffUuid, weekEnding, staffName, data, { activities: createdActivities, notes: createdNotes }); appendToTimesheetLog(data, weekEnding, staffName); } catch(sheetErr) { errors.push("Sheet error: " + sheetErr.message); } return jsonOut({ success: errors.length === 0, submitted: submitted, errors: errors, message: submitted.length + " job(s) submitted to ServiceM8." }); } catch(err) { return jsonOut({ success: false, message: "Server error: " + err.message }); } } // ---------------------------------------------------------------- // Test functions // ---------------------------------------------------------------- function testSheet() { Logger.log(SpreadsheetApp.openById(SHEET_ID).getName()); } function testRoles() { sm8Get("securityrole.json").forEach(function(r) { Logger.log(r.name + " = " + r.uuid); }); } function findStaff() { sm8Get("staff.json").forEach(function(s) { Logger.log(s.first + " " + s.last + " = " + s.uuid); }); }
